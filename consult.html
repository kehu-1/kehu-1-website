document.getElementById('consult-form').addEventListener('submit', function(e) { e.preventDefault(); const submitBtn = this.querySelector('.submit-btn'); const originalText = submitBtn.innerText; submitBtn.innerText = "发送中..."; submitBtn.disabled = true; const formData = new FormData(this); const dataObj = {}; // 把数据转成普通对象，过滤掉空值 formData.forEach((value, key) => { if (value && value.trim() !== "") { dataObj[key] = value; } }); // 🚨 这里填入你真实的 Access Key dataObj["access_key"] = "a1b2c3d4-e5f6-7890-abcd-ef1234567890"; dataObj["subject"] = "官网新表单：咨询制造专家"; dataObj["from_name"] = "你的品牌名"; // 用 JSON 格式发送 fetch("https://api.web3forms.com/submit", { method: "POST", headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' }, body: JSON.stringify(dataObj) }) .then(response => { return response.json(); }) .then(data => { if (data.success) { alert('提交成功！我们的团队会尽快与您联系。'); this.reset(); } else { // 如果还是报错，这里会弹出具体原因 console.error("Web3Forms报错详情:", data); alert('发送失败: ' + data.message); } }) .catch(error => { console.error('请求失败:', error); alert('网络错误，请稍后再试。'); }) .finally(() => { submitBtn.innerText = originalText; submitBtn.disabled = false; }); });
